cmake_minimum_required(VERSION 3.24)
project(MegaFMM LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES OFF)
find_package(CUDAToolkit 12.4 REQUIRED)
add_library(megafmm STATIC solution/plan.cpp solution/scale_output.cu solution/sw_pipeline.cu solution/sw_multisize.cu solution/dense_hopper.cu solution/tuning_check.cu solution/dense_padding.cu)
set(MEGAFMM_CUTLASS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/cutlass" CACHE PATH "CUTLASS headers for Hopper primitives")
if(NOT EXISTS "${MEGAFMM_CUTLASS_DIR}/include/cute/tensor.hpp")
  message(FATAL_ERROR "SW v1 needs CUTLASS 3.5.1 headers; set MEGAFMM_CUTLASS_DIR")
endif()
target_include_directories(megafmm PUBLIC include PRIVATE solution "${MEGAFMM_CUTLASS_DIR}/include")
target_link_libraries(megafmm PUBLIC CUDA::cudart CUDA::cublasLt CUDA::cublas CUDA::cuda_driver)
target_compile_options(megafmm PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr> $<$<COMPILE_LANGUAGE:CUDA>:-gencode=arch=compute_90a,code=sm_90a>)
add_executable(megafmm_bench bench/benchmark.cu)
target_link_libraries(megafmm_bench PRIVATE megafmm)
target_compile_options(megafmm_bench PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-gencode=arch=compute_90a,code=sm_90a>)
add_executable(test_correctness tests/test_correctness.cu)
target_link_libraries(test_correctness PRIVATE megafmm)
target_compile_options(test_correctness PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-gencode=arch=compute_90a,code=sm_90a>)
add_executable(test_sw1024 tests/test_sw1024.cu)
target_link_libraries(test_sw1024 PRIVATE megafmm)
target_compile_options(test_sw1024 PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-gencode=arch=compute_90a,code=sm_90a>)
enable_testing()
add_test(NAME gpu_correctness COMMAND test_correctness)
add_test(NAME sw_1024_correctness COMMAND test_sw1024)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME exact_winograd_dag COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_dag.py")
endif()

add_executable(test_multisize tests/test_multisize.cu)
target_link_libraries(test_multisize PRIVATE megafmm)
target_compile_options(test_multisize PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-gencode=arch=compute_90a,code=sm_90a>)
add_test(NAME sw_multisize_implementation COMMAND test_multisize)

add_test(NAME tuned_correctness COMMAND test_correctness --tuned)
